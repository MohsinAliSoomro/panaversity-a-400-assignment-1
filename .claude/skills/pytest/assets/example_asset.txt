## Fixture Template
```python
import pytest


@pytest.fixture
def sample_data():
    """Provide sample data for tests."""
    return {
        'key1': 'value1',
        'key2': 'value2'
    }


def test_with_fixture(sample_data):
    """Test that uses a fixture."""
    assert sample_data['key1'] == 'value1'
```

## Parametrized Test Template
```python
import pytest


@pytest.mark.parametrize("input,expected", [
    (1, 2),
    (2, 4),
    (3, 6),
])
def test_doubled(input, expected):
    """Test with multiple input/expected pairs."""
    result = input * 2
    assert result == expected
```
